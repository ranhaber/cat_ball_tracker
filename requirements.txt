# Cat Dome - Detection & Tracking System
# Requirements for Raspberry Pi Zero 2W
#
# INSTALLATION ON RASPBERRY PI:
# =============================
# Most packages should be installed via apt (system-wide):
#   sudo apt install python3-numpy python3-opencv python3-flask python3-scipy python3-pillow python3-picamera2
#
# Then create venv with system packages:
#   python3 -m venv --system-site-packages venv
#   source venv/bin/activate
#
# For TFLite on RPi, use Google's wheel:
#   pip install --extra-index-url https://google-coral.github.io/py-repo/ tflite_runtime
#
# Or skip TFLite - system will use mock detections for testing

# WSGI server for production
gunicorn==21.2.0

# Fast JPEG encoding via libjpeg-turbo (NEON SIMD on ARM) — ~50-70% faster than cv2.imencode
# Falls back to cv2.imencode gracefully if not installed
simplejpeg

# WebSocket support for H.264 hardware streaming (v3.10.0+)
# Falls back to MJPEG streaming gracefully if not installed
flask-sock

# Below packages are usually pre-installed via apt on RPi
